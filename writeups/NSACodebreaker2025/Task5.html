    The solution:

    bruteforceUSB.py:

    import subprocess
import itertools
import string

character_list = string.ascii_letters + string.digits + string.punctuation

known_prefix = "*+$+8P{#U0Vw+&v|"

for chars in itertools.product(character_list, repeat=2):
    attempt = known_prefix + "".join(chars)
    try:
        # Call gocryptfs with the current password attempt (after installing)
        process = subprocess.run(
            ["gocryptfs", "-passwd", "/mnt/.data"],
            input=attempt.encode(),
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE
        )
        print(attempt)
		# Output of the original decryption attempt that was in red at the start
        if b"Password incorrect" not in process.stderr:
            print(f"Password found: {attempt}")
            break
    except KeyboardInterrupt:
        print("Brute force interrupted.")
        break
