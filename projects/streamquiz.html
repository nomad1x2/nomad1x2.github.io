import streamlit as st import json import random import time import os from pathlib import Path st.set_page_config( page_title="Quiz", layout="centered", initial_sidebar_state="collapsed", ) st.markdown(''' ''', unsafe_allow_html=True) # ── Helpers ─────────────────────────────────────────────────────────────────── def load_questions(json_path: str) -> dict: with open(json_path, "r", encoding="utf-8") as f: data = json.load(f) return data["questions"] def format_time(seconds: int) -> str: m, s = divmod(max(0, seconds), 60) return f"{m:02d}:{s:02d}" def get_rank(pct: float) -> str: if pct >= 90: return "Excellent" elif pct >= 80: return "Good" elif pct >= 70: return "Satisfactory" elif pct >= 60: return "Marginal" else: return "Needs improvement" def reset_quiz(): for key in list(st.session_state.keys()): del st.session_state[key] # ── Locate default quiz.json ────────────────────────────────────────────────── SCRIPT_DIR = Path(__file__).parent DEFAULT_JSON = "quiz.json" for candidate in [SCRIPT_DIR / "quiz.json", Path("quiz.json")]: if candidate.exists(): DEFAULT_JSON = str(candidate) break if "quiz_active" not in st.session_state: st.session_state.quiz_active = False # ══════════════════════════════════════════════════════════════════════════════ # CONFIG SCREEN # ══════════════════════════════════════════════════════════════════════════════ if not st.session_state.quiz_active: st.markdown("## Quiz Setup") json_path = st.text_input("Question bank (JSON file)", value=DEFAULT_JSON) col1, col2 = st.columns(2) with col1: num_questions = st.slider("Number of questions", 5, 100, 10, 5) #start, end, default, inc with col2: randomize = st.checkbox("Randomize order", value=True) st.markdown("---") col3, col4 = st.columns(2) with col3: use_timer = st.checkbox("Enable timer", value=False) with col4: timer_minutes = st.slider("Duration (minutes)", 1, 60, 10, disabled=not use_timer) instant_feedback = st.checkbox("Show correct answer after each question", value=True) st.markdown("") if st.button("Start Quiz"): if not os.path.exists(json_path): st.error(f"File not found: {json_path}") else: try: all_q = load_questions(json_path) keys = list(all_q.keys()) if randomize: random.shuffle(keys) chosen = keys[:min(num_questions, len(keys))] st.session_state.questions = {k: all_q[k] for k in chosen} st.session_state.q_keys = chosen st.session_state.quiz_active = True st.session_state.current_q = 0 st.session_state.selected_answers = {} st.session_state.answered = {} st.session_state.score = 0 st.session_state.start_time = time.time() st.session_state.q_start_time = time.time() st.session_state.elapsed_times = [] st.session_state.time_up = False st.session_state.use_timer = use_timer st.session_state.timer_seconds = timer_minutes * 60 st.session_state.instant_feedback = instant_feedback st.session_state.total = len(chosen) st.rerun() except Exception as e: st.error(f"Failed to load questions: {e}") # ══════════════════════════════════════════════════════════════════════════════ # QUIZ SCREEN # ══════════════════════════════════════════════════════════════════════════════ elif st.session_state.quiz_active: questions = st.session_state.questions q_keys = st.session_state.q_keys total = st.session_state.total idx = st.session_state.current_q # Timer check elapsed = int(time.time() - st.session_state.start_time) if st.session_state.use_timer: remaining = st.session_state.timer_seconds - elapsed if remaining <= 0 and not st.session_state.time_up: st.session_state.time_up = True st.session_state.current_q = total st.rerun() # ── RESULTS ────────────────────────────────────────────────────────────── if idx >= total: score = st.session_state.score pct = (score / total) * 100 if total else 0 rank = get_rank(pct) elapsed = int(time.time() - st.session_state.start_time) avg_t = ( sum(st.session_state.elapsed_times) / len(st.session_state.elapsed_times) if st.session_state.elapsed_times else 0 ) if st.session_state.get("time_up"): st.warning("Time expired.") st.markdown(f'
{score} / {total}
', unsafe_allow_html=True) st.markdown(f'
{pct:.0f}% — {rank}
', unsafe_allow_html=True) st.markdown(f'''
{score}
Correct
{total - score}
Incorrect
{format_time(elapsed)}
Total time
{avg_t:.1f}s
Avg / question
''', unsafe_allow_html=True) missed = [k for k in q_keys if k in st.session_state.answered and not st.session_state.answered[k]] if missed: with st.expander(f"Review missed questions ({len(missed)})"): for k in missed: q = questions[k] correct_key = list(q["answer"].keys())[0] correct_text = list(q["answer"].values())[0] your_raw = st.session_state.selected_answers.get(k, "Not answered") st.markdown(f"**{q['question']}**") st.markdown(f"Your answer: {your_raw}", unsafe_allow_html=True) st.markdown(f"Correct: {correct_key} {correct_text}", unsafe_allow_html=True) st.markdown("---") st.markdown("") c1, c2 = st.columns(2) with c1: if st.button("Restart Quiz"): reset_quiz() st.rerun() with c2: if st.button("New Settings"): reset_quiz() st.rerun() # ── ACTIVE QUESTION ─────────────────────────────────────────────────────── else: q_key = q_keys[idx] q_data = questions[q_key] opts = q_data["options"] correct_key = list(q_data["answer"].keys())[0] already_answered = q_key in st.session_state.answered # Build timer string if needed timer_str = "" if st.session_state.use_timer: remaining = st.session_state.timer_seconds - elapsed warn_cls = "timer-warn" if remaining < 60 else "timer-text" timer_str = f'{format_time(remaining)}' # ── Progress label + Exit button on same row ── prog_col, exit_col = st.columns([8, 1]) with prog_col: timer_suffix = f"  {timer_str}" if timer_str else "" st.markdown( f'
' f'Question {idx + 1} of {total}  ·  {st.session_state.score} correct{timer_suffix}' f'
', unsafe_allow_html=True, ) with exit_col: if st.button("Exit", key="exit_btn", use_container_width=True): st.session_state.current_q = total st.rerun() # ── Progress bar full-width below ── st.progress(idx / total) st.markdown("
", unsafe_allow_html=True) # Question card st.markdown( f'
' f'
Question {idx + 1}
' f'
{q_data["question"]}
' f'
', unsafe_allow_html=True, ) option_labels = [f"{k} {v}" for k, v in opts.items()] # Restore saved selection default_idx = None if q_key in st.session_state.selected_answers: saved = st.session_state.selected_answers[q_key] for i, lbl in enumerate(option_labels): if lbl == saved: default_idx = i break # Answers (radio) selected = st.radio( "Select an answer:", option_labels, index=default_idx, key=f"radio_{q_key}", disabled=already_answered, ) # Next / Finish button — full width, below answers btn_label = "Finish" if (already_answered and idx == total - 1) else "Next" if st.button(btn_label, use_container_width=True): if not already_answered: radio_val = st.session_state.get(f"radio_{q_key}") if radio_val: chosen_key = None for ok, ov in opts.items(): if f"{ok} {ov}" == radio_val: chosen_key = ok break if chosen_key: st.session_state.selected_answers[q_key] = radio_val is_correct = chosen_key == correct_key st.session_state.answered[q_key] = is_correct if is_correct: st.session_state.score += 1 st.session_state.elapsed_times.append( time.time() - st.session_state.q_start_time ) if not st.session_state.instant_feedback: st.session_state.current_q += 1 st.session_state.q_start_time = time.time() st.rerun() else: st.session_state.current_q += 1 st.session_state.q_start_time = time.time() st.rerun() # Feedback if already_answered and st.session_state.instant_feedback: is_correct = st.session_state.answered[q_key] expl = list(q_data["answer"].values())[0] if is_correct: st.markdown( f'
Correct' f'
{expl}
', unsafe_allow_html=True, ) else: correct_display = f"{correct_key} {opts.get(correct_key, '')}" st.markdown( f'
Incorrect — correct answer: {correct_display}' f'
{expl}
', unsafe_allow_html=True, ) # Auto-refresh for live timer if st.session_state.use_timer and not already_answered: time.sleep(0.5) st.rerun()
